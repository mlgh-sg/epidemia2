epidemia

epidemia — Bayesian semi-mechanistic modelling of infectious diseases.

A Python counterpart of the R package epidemia. Latent daily infections are propagated by a renewal process; the reproduction number and observations are modelled with NumPyro, and inference uses a fast NUTS backend (nutpie by default, or BlackJAX / NumPyro).

def expected_observations(infections, i2o, ascertainment=1.0): 
 77def expected_observations(infections, i2o, ascertainment=1.0):
 78    """Expected observations from infections via an infection-to-observation delay.
 79
 80    Parameters
 81    ----------
 82    infections : array (N,)
 83        Latent daily infections.
 84    i2o : array (K,)
 85        Infection-to-observation delay distribution; ``i2o[k]`` weights
 86        infections ``k`` days before the observation. Need not sum to one (it
 87        can also encode an overall ascertainment scale).
 88    ascertainment : float or array (N,)
 89        Time-varying multiplier (e.g. an ascertainment/IFR rate).
 90
 91    Returns
 92    -------
 93    y : array (N,)
 94        Expected observations on each day.
 95    """
 96    infections = jnp.asarray(infections)
 97    i2o = jnp.asarray(i2o)
 98    N = infections.shape[0]
 99    conv = jnp.convolve(infections, i2o)[:N]  # causal: sum_k i2o[k] * i_{t-k}
100    return jnp.asarray(ascertainment) * conv

def infectiousness(infections, gen): 
67def infectiousness(infections, gen):
68    """Total infectiousness ``sum_{s} i_{t-s} * gen_s`` (the renewal weight)."""
69    infections = jnp.asarray(infections)
70    gen = jnp.asarray(gen)
71    N = infections.shape[0]
72    # shift by one day so day t uses only past infections
73    shifted = jnp.concatenate([jnp.zeros(1, infections.dtype), infections[:-1]])

def random_walk(scale, noise, intercept): 
103def random_walk(scale, noise, intercept):
104    """Build a random-walk linear predictor ``intercept + cumsum(scale * noise)``.
105
106    ``noise`` are standardised (unit-normal) increments — a *non-centred*
107    parameterisation, which gives the sampler a far easier geometry than
108    drawing the increments directly at the unknown ``scale``.
109    """
110    steps = jnp.asarray(scale) * jnp.asarray(noise)
111    return jnp.asarray(intercept) + jnp.cumsum(steps)

def renewal_infections(R, seeds, gen): 
25def renewal_infections(R, seeds, gen):
26    """Propagate infections via the discrete renewal equation.
27
28    Parameters
29    ----------
30    R : array (N,)
31        Reproduction number on each of the ``N`` modelled days. Values before
32        the end of the seeding window are ignored.
33    seeds : array (v,)
34        Seeded infections for the first ``v`` days (parameters of the model).
35    gen : array (L,)
36        Generation-interval PMF; ``gen[s-1]`` is the probability that a
37        secondary infection occurs ``s`` days after the primary one.
38
39    Returns
40    -------
41    infections : array (N,)
42        Latent daily infections, with the first ``v`` entries equal to ``seeds``.
43    """
44    R = jnp.asarray(R)
45    seeds = jnp.asarray(seeds)
46    gen = jnp.asarray(gen)
47    L = gen.shape[0]
48    v = seeds.shape[0]
49
50    # Buffer of the most recent ``L`` infections, most-recent-first, so that
51    # dot(buffer, gen) == sum_{s=1..L} i_{t-s} * gen[s-1].
52    rev = seeds[::-1]
53    if v >= L:
54        buf0 = rev[:L]
55    else:
56        buf0 = jnp.concatenate([rev, jnp.zeros(L - v, dtype=seeds.dtype)])
57
58    def step(buf, R_t):
59        i_t = R_t * jnp.dot(buf, gen)
60        new_buf = jnp.concatenate([i_t[None], buf[:-1]])
61        return new_buf, i_t
62
63    _, infections_after_seed = jax.lax.scan(step, buf0, R[v:])
64    return jnp.concatenate([seeds, infections_after_seed])

Propagate infections via the discrete renewal equation.

  • seeds (array (v,)): Seeded infections for the first v days (parameters of the model).
  • gen (array (L,)): Generation-interval PMF; gen[s-1] is the probability that a secondary infection occurs s days after the primary one.

def renewal_model(y, config: EpiConfig): 
 75def renewal_model(y, config: EpiConfig):
 76    """NumPyro model for a single-population renewal process.
 77
 78    Parameters
 79    ----------
 80    y : array (N,)
 81        Observed series. Use ``nan`` for days that are not observed (e.g. the
 82        seeding period); those days are masked out of the likelihood.
 83    config : EpiConfig
 84        Model configuration.
 85    """
 86    y = jnp.asarray(y, dtype=jnp.float32)
 87    N = y.shape[0]
 88
 89    # the random-walk index is model structure (fixed), so resolve its size
 90    # statically from NumPy rather than from a traced JAX array.
 91    rw_index_np = np.arange(N) if config.rw_index is None else np.asarray(config.rw_index)
 92    n_steps = int(rw_index_np.max()) + 1
 93    rw_index = jnp.asarray(rw_index_np)
 94
 95    # --- priors ------------------------------------------------------------
 96    intercept = numpyro.sample(
 97        "intercept", dist.Normal(config.intercept_loc, config.intercept_scale)
 98    )
 99    rw_scale = numpyro.sample("rw_scale", dist.HalfNormal(config.rw_prior_scale))
100    rw_noise = numpyro.sample("rw_noise", dist.Normal(0.0, 1.0).expand([n_steps]))
101    seed = numpyro.sample("seed", dist.Exponential(1.0 / config.seed_prior_mean))
102
103    # --- transmission: R_t = g^{-1}(intercept + random walk) ---------------
104    walk = random_walk(rw_scale, rw_noise, 0.0)   # cumulative, non-centred
105    eta = intercept + walk[rw_index]
106    R = link_inv(eta, config.link)
107    numpyro.deterministic("Rt", R)
108
109    # --- infections via the renewal equation -------------------------------
110    seeds = jnp.full((config.seed_days,), seed)
111    infections = renewal_infections(R, seeds, config.gen)
112    numpyro.deterministic("infections", infections)
113
114    # --- expected observations ---------------------------------------------
115    E = expected_observations(infections, config.i2o, 1.0) + 1e-6
116    numpyro.deterministic("E_obs", E)
117
118    # --- likelihood (masking unobserved days) ------------------------------
119    mask = ~jnp.isnan(y)
120    y_safe = jnp.where(mask, y, 0.0)
121    if config.family == "poisson":
122        obs_dist = dist.Poisson(E)
123    elif config.family == "neg_binom":
124        phi = numpyro.sample("reciprocal_dispersion", dist.HalfNormal(5.0))
125        obs_dist = dist.NegativeBinomial2(E, phi)
126    else:
127        raise ValueError(f"unknown family: {config.family!r}")
128
129    with numpyro.handlers.mask(mask=mask):
130        numpyro.sample("y", obs_dist, obs=y_safe)
